###############################################################
# Forecast Accuracy Function
###############################################################

forecast_metrics <- function(
    obs,
    pred
) {

    obs <- as.numeric(
        obs
    )

    pred <- as.numeric(
        pred
    )

    pred <- pmax(
        pred,
        0
    )

    data.frame(

        RMSE =
            sqrt(
                mean(
                    (
                        obs -
                        pred
                    )^2,
                    na.rm = TRUE
                )
            ),

        MAE =
            mean(
                abs(
                    obs -
                    pred
                ),
                na.rm = TRUE
            ),

        MAPE =
            mean(

                abs(
                    obs -
                    pred
                ) /

                    pmax(
                        abs(obs),
                        1
                    ),

                na.rm = TRUE

            ) * 100,

        WAPE =
            sum(
                abs(
                    obs -
                    pred
                ),
                na.rm = TRUE
            ) /

            pmax(
                sum(
                    abs(obs),
                    na.rm = TRUE
                ),
                1
            )

    )

}


###############################################################
# Forecast Each Store
###############################################################

for (
    i in seq_len(
        N_STORES
    )
) {

    cat(
        "\nForecasting Store:",
        selected_stores[i],
        "\n"
    )


    ###########################################################
    # Time Series Construction
    ###########################################################

    y <- ts(

        sales_matrix[, i],

        frequency = 7

    )


    ###########################################################
    # Train/Test Split
    ###########################################################

    train <- ts(

        y[
            1:train_length
        ],

        frequency = 7

    )


    test <- ts(

        y[
            (train_length + 1):
                length(y)
        ],

        frequency = 7

    )


    ###########################################################
    # Forecast Models
    ###########################################################

    fit_arima <-
        tryCatch(

            auto.arima(
                train
            ),

            error =
                function(e) {

                    cat(
                        "ARIMA failed for Store ",
                        selected_stores[i],
                        ": ",
                        conditionMessage(e),
                        "\n",
                        sep = ""
                    )

                    NULL

                }

        )


    fit_ets <-
        tryCatch(

            ets(
                train
            ),

            error =
                function(e) {

                    cat(
                        "ETS failed for Store ",
                        selected_stores[i],
                        ": ",
                        conditionMessage(e),
                        "\n",
                        sep = ""
                    )

                    NULL

                }

        )


    fit_tbats <-
        tryCatch(

            tbats(
                train
            ),

            error =
                function(e) {

                    cat(
                        "TBATS failed for Store ",
                        selected_stores[i],
                        ": ",
                        conditionMessage(e),
                        "\n",
                        sep = ""
                    )

                    NULL

                }

        )


    fit_theta <-
        tryCatch(

            thetaf(
                train,
                h = forecast_horizon
            ),

            error =
                function(e) {

                    cat(
                        "Theta failed for Store ",
                        selected_stores[i],
                        ": ",
                        conditionMessage(e),
                        "\n",
                        sep = ""
                    )

                    NULL

                }

        )


    ###########################################################
    # Forecast Generation
    ###########################################################

    pred_arima <-
        if (!is.null(fit_arima)) {

            as.numeric(
                forecast(
                    fit_arima,
                    h = forecast_horizon
                )$mean
            )

        } else {

            rep(
                mean(train),
                forecast_horizon
            )

        }


    pred_ets <-
        if (!is.null(fit_ets)) {

            as.numeric(
                forecast(
                    fit_ets,
                    h = forecast_horizon
                )$mean
            )

        } else {

            rep(
                mean(train),
                forecast_horizon
            )

        }


    pred_tbats <-
        if (!is.null(fit_tbats)) {

            as.numeric(
                forecast(
                    fit_tbats,
                    h = forecast_horizon
                )$mean
            )

        } else {

            rep(
                mean(train),
                forecast_horizon
            )

        }


    pred_theta <-
        if (!is.null(fit_theta)) {

            as.numeric(
                fit_theta$mean
            )

        } else {

            rep(
                mean(train),
                forecast_horizon
            )

        }


    ###########################################################
    # Remove Negative Demand
    ###########################################################

    pred_arima <-
        pmax(
            pred_arima,
            0
        )

    pred_ets <-
        pmax(
            pred_ets,
            0
        )

    pred_tbats <-
        pmax(
            pred_tbats,
            0
        )

    pred_theta <-
        pmax(
            pred_theta,
            0
        )


    ###########################################################
    # Forecast Evaluation
    ###########################################################

    store_results <- rbind(

        cbind(
            Model = "ARIMA",
            forecast_metrics(
                test,
                pred_arima
            )
        ),

        cbind(
            Model = "ETS",
            forecast_metrics(
                test,
                pred_ets
            )
        ),

        cbind(
            Model = "TBATS",
            forecast_metrics(
                test,
                pred_tbats
            )
        ),

        cbind(
            Model = "Theta",
            forecast_metrics(
                test,
                pred_theta
            )
        )

    )


    store_results$Store <-
        selected_stores[i]


    forecast_accuracy <-
        rbind(
            forecast_accuracy,
            store_results
        )


###########################################################
# Save Models
###########################################################

forecast_models[[selected_stores[i]]] <-

    list(

        ARIMA = fit_arima,

        ETS = fit_ets,

        TBATS = fit_tbats,

        Theta = fit_theta

    )

    ###########################################################
    # Demand Parameters for SAR-Q
    ###########################################################

    demand_base[i] <-
        mean(
            train,
            na.rm = TRUE
        )


    demand_sd[i] <-
        max(

            sd(
                train,
                na.rm = TRUE
            ),

            MIN_DEMAND_SD

        )


    ###########################################################
    # ARIMA Residuals
    #
    # Input for Copula Dependence Modeling
    ###########################################################

    if (
        !is.null(
            fit_arima
        )
    ) {

        residuals_i <-
            as.numeric(
                residuals(
                    fit_arima
                )
            )

        residual_length <-
            min(
                length(residuals_i),
                train_length
            )

        demand_residuals[
            seq_len(
                residual_length
            ),
            i
        ] <-
            residuals_i[
                seq_len(
                    residual_length
                )
            ]

    }

}

# Helper function for extracting log-likelihood
get_logLik <- function(model) {
    
    if (is.null(model)) {
        
        return(NA_real_)
        
    }
    
    tryCatch(
        
        as.numeric(
            logLik(model)
        ),
        
        error = function(e) {
            
            NA_real_
        }
    )
}


# Helper function for extracting AIC
get_AIC <- function(model) {
    
    if (is.null(model)) {
        
        return(NA_real_)
        
    }
    
    tryCatch(
        
        AIC(model),
        
        error = function(e) {
            
            NA_real_
        }
    )
}

###############################################################
# Epsilon Schedule
###############################################################


epsilon_schedule <- function(

    episode

){

    eps <-

        EPSILON_START *

        (EPSILON_DECAY ^ (episode - 1))


    max(

        EPSILON_END,

        eps

    )

}



###############################################################
# Matrix Normalization
#
# Row-normalized matrices are used for spatial/cross-store
# information. This also protects against scale differences
# between W and copula dependence matrices.
###############################################################


row_normalize_matrix <- function(

    M

){

    M <-

        as.matrix(M)


    row_sums <-

        rowSums(

            abs(M)

        )


    result <-

        matrix(

            0,

            nrow = nrow(M),

            ncol = ncol(M)

        )


    valid_rows <-

        row_sums > EPS


    result[valid_rows, ] <-

        M[valid_rows, ] /

        row_sums[valid_rows]


    result

}

###############################################################
# Receive Shipments
###############################################################


receive_shipments <- function(env){

    ###########################################################
    # Arrivals scheduled for this period
    ###########################################################

    arrivals <-

        env$pipeline[, 1]


    ###########################################################
    # Add arrivals to inventory
    ###########################################################

    env$inventory <-

        env$inventory +

        arrivals


    ###########################################################
    # Enforce capacity
    ###########################################################

    env$inventory <-

        pmin(

            pmax(

                env$inventory,

                MIN_INVENTORY

            ),

            MAX_CAPACITY

        )


    ###########################################################
    # Shift pipeline
    ###########################################################

    if(MAX_LEAD_TIME > 1){

        env$pipeline[, 1:(MAX_LEAD_TIME - 1)] <-

            env$pipeline[, 2:MAX_LEAD_TIME]

    }


    ###########################################################
    # Clear final pipeline position
    ###########################################################

    env$pipeline[, MAX_LEAD_TIME] <-

        0


    return(env)

}

    # Validate actions
    ###########################################################

    if(length(actions) != N_STORES){

        stop(
            "Number of actions must equal N_STORES."
        )

    }


    if(any(

        !is.finite(actions) |

        actions < 1 |

        actions > N_ACTIONS |

        actions != floor(actions)

    )){

        stop(
            "Invalid action index detected."
        )

    }


    ###########################################################
    # Convert action indices to order quantities
    ###########################################################

    order_qty <-

        ACTION_SPACE[actions]


    ###########################################################
    # Place orders according to actual lead time
    ###########################################################

    env$pipeline[, LEAD_TIME] <-

        env$pipeline[, LEAD_TIME] +

        order_qty


    ###########################################################
    # Record current orders
    ###########################################################

    env$orders <-

        order_qty


    return(env)

}



###############################################################
# Network Demand Generator
###############################################################


generate_network_demand <- function(day){

    ###########################################################
    # Boundary condition
    ###########################################################

    if(day > N_DAYS){

        return(

            rep(

                0,

                N_STORES

            )

        )

    }


    ###########################################################
    # Demand comes from FILE 02
    #
    # sales_matrix is treated as the realized network demand
    # trajectory generated in FILE 02.
    ###########################################################

    demand <-

        as.numeric(

            sales_matrix[day, ]

        )


    ###########################################################
    # Demand validation
    ###########################################################

    if(length(demand) != N_STORES){

        stop(
            "Generated demand has incorrect dimension."
        )

    }


    if(any(!is.finite(demand))){

        stop(
            "Generated demand contains NA/NaN/Inf."
        )

    }


    ###########################################################
    # Demand cannot be negative
    ###########################################################

    demand <-

        pmax(

            demand,

            0

        )


    return(demand)

}



###############################################################
# Inventory Balance
###############################################################


update_inventory <- function(

    env,

    demand

){

    ###########################################################
    # Inventory available before demand
    ###########################################################

    available_inventory <-

        pmin(

            pmax(

                env$inventory,

                MIN_INVENTORY

            ),

            MAX_CAPACITY

        )


    ###########################################################
    # Actual sales
    ###########################################################

    sales <-

        pmin(

            available_inventory,

            pmax(

                demand,

                0

            )

        )


    ###########################################################
    # Unmet demand
    ###########################################################

    shortage <-

        pmax(

            demand - sales,

            0

        )


    ###########################################################
    # Inventory after sales
    ###########################################################

    ending_inventory <-

        available_inventory -

        sales


    ###########################################################
    # Enforce inventory bounds
    ###########################################################

    ending_inventory <-

        pmin(

            pmax(

                ending_inventory,

                MIN_INVENTORY

            ),

            MAX_CAPACITY

        )


    ###########################################################
    # Update environment
    ###########################################################

    env$inventory <-

        ending_inventory


    env$demand <-

        pmax(

            demand,

            0

        )


    env$sales <-

        sales


    env$shortage <-

        shortage


    ###########################################################
    # Cumulative stockout count
    ###########################################################

    env$stockout <-

        env$stockout +

        as.numeric(

            shortage > 0

        )


    return(env)

}



###############################################################
# Cost and Reward Calculation
###############################################################


calculate_reward <- function(

    env,

    shortage

){

    ###########################################################
    # Holding cost
    #
    # Charged on ending inventory.
    ###########################################################

    holding_cost <-

        HOLDING_COST *

        pmax(

            env$inventory,

            0

        )


    ###########################################################
    # Shortage cost
    ###########################################################

    shortage_cost <-

        SHORTAGE_COST *

        pmax(

            shortage,

            0

        )


    ###########################################################
    # Ordering cost
    #
    # Fixed/variable ordering cost based on order quantity.
    ###########################################################

    ordering_cost <-

        ORDERING_COST *

        pmax(

            env$orders,

            0

        )


    ###########################################################
    # Purchasing cost
    ###########################################################

    purchasing_cost <-

        PURCHASE_COST *

        pmax(

            env$orders,

            0

        )


    ###########################################################
    # Total cost
    ###########################################################

    total_cost <-

        holding_cost +

        shortage_cost +

        ordering_cost +

        purchasing_cost


    ###########################################################
    # RL reward
    #
    # Reward is negative inventory cost.
    ###########################################################

    reward <-

        -REWARD_SCALE *

        total_cost


    ###########################################################
    # Save daily components
    ###########################################################

    env$holding_cost <-

        holding_cost


    env$shortage_cost <-

        shortage_cost


    env$ordering_cost <-

        ordering_cost


    env$purchasing_cost <-

        purchasing_cost


    ###########################################################
    # Cumulative performance
    ###########################################################

    env$cost <-

        env$cost +

        total_cost


    env$reward <-

        env$reward +

        reward


    ###########################################################
    # Return
    ###########################################################

    list(

        env = env,

        reward = reward,

        cost = total_cost,

        holding_cost = holding_cost,

        shortage_cost = shortage_cost,

        ordering_cost = ordering_cost,

        purchasing_cost = purchasing_cost

    )

}



###############################################################
# Environment Transition
###############################################################


environment_step <- function(

    env,

    actions

){

    ###########################################################
    # Validate current environment
    ###########################################################

    if(env$day > N_DAYS){

        stop(
            "environment_step called after episode termination."
        )

    }


    ###########################################################
    # Save inventory before transition
    ###########################################################

    inventory_before <-

        env$inventory


    ###########################################################
    # Receive previous orders
    ###########################################################

    env <-

        receive_shipments(

            env

        )


    ###########################################################
    # Place new orders
    ###########################################################

    env <-

        place_orders(

            env,

            actions

        )


    ###########################################################
    # Demand realization
    ###########################################################

    demand <-

        generate_network_demand(

            env$day

        )


    ###########################################################
    # Inventory update
    ###########################################################

    env <-

        update_inventory(

            env,

            demand

        )


    ###########################################################
    # Shortage
    ###########################################################

    shortage <-

        env$shortage


    ###########################################################
    # Reward and cost
    ###########################################################

    reward_result <-

        calculate_reward(

            env,

            shortage

        )


    env <-

        reward_result$env


    reward <-

        reward_result$reward


    cost <-

        reward_result$cost


    ###########################################################
    # Save daily histories
    ###########################################################

    env$inventory_history <-

        rbind(

            env$inventory_history,

            env$inventory

        )


    env$demand_history <-

        rbind(

            env$demand_history,

            demand

        )


    env$sales_history <-

        rbind(

            env$sales_history,

            env$sales

        )


    env$shortage_history <-

        rbind(

            env$shortage_history,

            shortage

        )


    env$reward_history <-

        rbind(

            env$reward_history,

            reward

        )


    env$cost_history <-

        rbind(

            env$cost_history,

            cost

        )


    env$holding_cost_history <-

        rbind(

            env$holding_cost_history,

            env$holding_cost

        )


    env$shortage_cost_history <-

        rbind(

            env$shortage_cost_history,

            env$shortage_cost

        )


    env$ordering_cost_history <-

        rbind(

            env$ordering_cost_history,

            env$ordering_cost

        )


    env$purchasing_cost_history <-

        rbind(

            env$purchasing_cost_history,

            env$purchasing_cost

        )


    env$stockout_history <-

        rbind(

            env$stockout_history,

            as.numeric(

                shortage > 0

            )

        )


    ###########################################################
    # Advance time
    ###########################################################

    env$day <-

        env$day + 1L


    ###########################################################
    # Episode termination
    ###########################################################

    done <-

        env$day > N_DAYS


    ###########################################################
    # Return
    ###########################################################

    list(

        env = env,

        reward = reward,

        cost = cost,

        demand = demand,

        sales = env$sales,

        shortage = shortage,

        inventory = env$inventory,

        inventory_before = inventory_before,

        orders = env$orders,

        holding_cost = env$holding_cost,

        shortage_cost = env$shortage_cost,

        ordering_cost = env$ordering_cost,

        purchasing_cost = env$purchasing_cost,

        done = done

    )

}

network_KPI <- function(env){

    ###########################################################
    # Total reward
    ###########################################################

    total_reward <-

        sum(

            env$reward_history,

            na.rm = TRUE

        )


    ###########################################################
    # Total cost
    ###########################################################

    total_cost <-

        sum(

            env$cost_history,

            na.rm = TRUE

        )


    ###########################################################
    # Cost components
    ###########################################################

    total_holding_cost <-

        sum(

            env$holding_cost_history,

            na.rm = TRUE

        )


    total_shortage_cost <-

        sum(

            env$shortage_cost_history,

            na.rm = TRUE

        )


    total_ordering_cost <-

        sum(

            env$ordering_cost_history,

            na.rm = TRUE

        )


    total_purchasing_cost <-

        sum(

            env$purchasing_cost_history,

            na.rm = TRUE

        )


    ###########################################################
    # Total demand
    ###########################################################

    total_demand <-

        sum(

            env$demand_history,

            na.rm = TRUE

        )


    ###########################################################
    # Total sales
    ###########################################################

    total_sales <-

        sum(

            env$sales_history,

            na.rm = TRUE

        )


    ###########################################################
    # Total shortage
    ###########################################################

    total_shortage <-

        sum(

            env$shortage_history,

            na.rm = TRUE

        )


    ###########################################################
    # Fill rate
    ###########################################################

    fill_rate <-

        ifelse(

            total_demand > 0,

            total_sales / total_demand,

            NA_real_

        )


    ###########################################################
    # Total stockouts
    ###########################################################

    total_stockouts <-

        sum(

            env$stockout_history,

            na.rm = TRUE

        )


    ###########################################################
    # Mean inventory
    ###########################################################

    mean_inventory <-

        mean(

            env$inventory_history,

            na.rm = TRUE

        )


    ###########################################################
    # Return KPI
    ###########################################################

    data.frame(

        TotalReward =
            total_reward,

        TotalCost =
            total_cost,

        HoldingCost =
            total_holding_cost,

        ShortageCost =
            total_shortage_cost,

        OrderingCost =
            total_ordering_cost,

        PurchasingCost =
            total_purchasing_cost,

        TotalDemand =
            total_demand,

        TotalSales =
            total_sales,

        TotalShortage =
            total_shortage,

        FillRate =
            fill_rate,

        TotalStockouts =
            total_stockouts,

        MeanInventory =
            mean_inventory

    )

}

###############################################################
# Safe Z-score Scaling
###############################################################


safe_scale <- function(x){

    x <- as.numeric(x)

    if(any(!is.finite(x))){

        stop(
            "safe_scale received non-finite values."
        )

    }

    s <- sd(x, na.rm = TRUE)

    if(

        is.na(s) ||

        s < EPS

    ){

        return(

            rep(

                0,

                length(x)

            )

        )

    }

    as.numeric(

        (x - mean(x, na.rm = TRUE)) / s

    )

}

###############################################################
# Demand Scaling
#
# Demand is represented relative to the store's baseline.
#
# demand_ratio = current demand / expected demand
#
# This avoids putting raw demand values directly into the
# state discretizer.
###############################################################


demand_ratio <- function(

    demand,

    store_id

){

    baseline <-

        max(

            as.numeric(

                demand_base[store_id]

            ),

            EPS

        )


    ratio <-

        as.numeric(demand) /

        baseline


    ratio <-

        ifelse(

            is.finite(ratio),

            ratio,

            0

        )


    ratio

}



###############################################################
# Spatial Demand Scaling
#
# Spatial demand is represented relative to the corresponding
# spatial baseline demand.
###############################################################


spatial_demand_ratio <- function(

    spatial_demand,

    store_id

){

    baseline <-

        max(

            abs(

                SpatialBaseline[store_id]

            ),

            EPS

        )


    ratio <-

        spatial_demand /

        baseline


    ratio <-

        ifelse(

            is.finite(ratio),

            ratio,

            0

        )


    ratio

}



###############################################################
# Construct Continuous Store State
###############################################################


construct_store_state <- function(

    env,

    store_id

){

    ###########################################################
    # Validate store
    ###########################################################

    if(

        store_id < 1 ||

        store_id > N_STORES

    ){

        stop(

            "Invalid store_id."

        )

    }


    ###########################################################
    # Current day
    ###########################################################

    day <-

        min(

            max(

                env$day,

                1

            ),

            N_DAYS

        )


    ###########################################################
    # Current observed demand
    ###########################################################

    current_demand <-

        as.numeric(

            sales_matrix[
                day,
                store_id
            ]

        )


    current_demand <-

        max(

            current_demand,

            0

        )


    ###########################################################
    # Current network demand
    ###########################################################

    network_demand <-

        env$demand


    ###########################################################
    # Spatial demand
    ###########################################################

    spatial_demand <-

        spatial_demand_feature(

            network_demand

        )[store_id]


    ###########################################################
    # Normalize current demand
    ###########################################################

    demand_state <-

        demand_ratio(

            current_demand,

            store_id

        )


    ###########################################################
    # Normalize spatial demand
    ###########################################################

    spatial_state <-

        spatial_demand_ratio(

            spatial_demand,

            store_id

        )


    ###########################################################
    # Inventory state
    #
    # Range approximately [0,1].
    ###########################################################

    inventory_state <-

        env$inventory[store_id] /

        MAX_CAPACITY


    inventory_state <-

        min(

            max(

                inventory_state,

                0

            ),

            1

        )


    ###########################################################
    # State vector
    ###########################################################

    state <-

        c(

            Inventory =
                inventory_state,

            Demand =
                demand_state,

            Forecast =
                ForecastFeature[store_id],

            Volatility =
                VolatilityFeature[store_id],

            SpatialDemand =
                spatial_state,

            CopulaDependence =
                CopulaFeature[store_id]

        )


    ###########################################################
    # Final validation
    ###########################################################

    if(

        length(state) != STATE_DIM

    ){

        stop(

            "Constructed state has incorrect dimension."

        )

    }


    if(

        any(!is.finite(state))

    ){

        stop(

            "Constructed state contains NA/NaN/Inf."

        )

    }


    return(state)

}



###############################################################
# State Discretization
#
# IMPORTANT:
#
# Different state variables have different natural scales.
# Therefore, DO NOT use one common -3 to +3 scale for every
# variable.
###############################################################


discretize_state <- function(

    state_vector

){

    if(

        length(state_vector) != STATE_DIM

    ){

        stop(

            "State vector has incorrect dimension."

        )

    }


    if(

        any(!is.finite(state_vector))

    ){

        stop(

            "State vector contains non-finite values."

        )

    }


    ###########################################################
    # Inventory
    #
    # 0 to 1
    ###########################################################

    inventory_bin <-

        findInterval(

            state_vector[1],

            c(

                0,

                0.20,

                0.40,

                0.60,

                0.80,

                1.00

            ),

            all.inside = TRUE

        )


    ###########################################################
    # Demand ratio
    #
    # 0 = no demand
    # 1 = baseline demand
    # >1 = above-baseline demand
    ###########################################################

    demand_bin <-

        findInterval(

            state_vector[2],

            c(

                0,

                0.50,

                0.80,

                1.20,

                1.50,

                2.00

            ),

            all.inside = TRUE

        )


    ###########################################################
    # Forecast z-score
    ###########################################################

    forecast_bin <-

        findInterval(

            state_vector[3],

            c(

                -Inf,

                -1.0,

                -0.5,

                0,

                0.5,

                1.0

            ),

            all.inside = TRUE

        )


    ###########################################################
    # Volatility z-score
    ###########################################################

    volatility_bin <-

        findInterval(

            state_vector[4],

            c(

                -Inf,

                -1.0,

                -0.5,

                0,

                0.5,

                1.0

            ),

            all.inside = TRUE

        )


    ###########################################################
    # Spatial demand ratio
    ###########################################################

    spatial_bin <-

        findInterval(

            state_vector[5],

            c(

                0,

                0.50,

                0.80,

                1.20,

                1.50,

                2.00

            ),

            all.inside = TRUE

        )


    ###########################################################
    # Copula dependence z-score
    ###########################################################

    copula_bin <-

        findInterval(

            state_vector[6],

            c(

                -Inf,

                -1.0,

                -0.5,

                0,

                0.5,

                1.0

            ),

            all.inside = TRUE

        )


    ###########################################################
    # Return discrete state
    ###########################################################

    bins <-

        c(

            inventory_bin,

            demand_bin,

            forecast_bin,

            volatility_bin,

            spatial_bin,

            copula_bin

        )


    as.integer(bins)

}



###############################################################
# State Key
###############################################################


state_key <- function(

    state_vector

){

    bins <-

        discretize_state(

###############################################################

###############################################################


diagnose_state_diversity <- function(

    env,

    n_days = min(30, N_DAYS)

){

    ###########################################################
    # Save original day
    ###########################################################

    original_day <-

        env$day


    keys <-

        character(0)


    ###########################################################
    # Examine multiple days
    ###########################################################

    for(day in seq_len(n_days)){

        env$day <- day

        env$demand <-

            as.numeric(

                sales_matrix[day, ]

            )


        for(i in seq_len(N_STORES)){

            s <-

                construct_store_state(

                    env,

                    i

                )


            keys <-

                c(

                    keys,

                    state_key(s)

                )

        }

    }


    ###########################################################
    # Restore day
    ###########################################################

    env$day <-

        original_day


    ###########################################################
    # Diagnostics
    ###########################################################

    unique_states <-

        length(

            unique(keys)

        )


    total_states <-

        length(keys)


    diversity_ratio <-

        unique_states /

        max(

            total_states,

            1

        )


    cat("\n")
    cat("========================================\n")
    cat("STATE DIVERSITY DIAGNOSTIC\n")
    cat("========================================\n")


    cat(
        "Total store-period states:",
        total_states,
        "\n"
    )


    cat(
        "Unique states:",
        unique_states,
        "\n"
    )


    cat(
        "State diversity ratio:",
        round(diversity_ratio, 4),
        "\n"
    )


    if(

        unique_states <= 1

    ){

        warning(

            "CRITICAL: All store-period observations map to one state."

        )

    }


    if(

        diversity_ratio < 0.05

    ){

        warning(

            "WARNING: State diversity is extremely low."

        )

    }


    invisible(

        list(

            total = total_states,

            unique = unique_states,

            ratio = diversity_ratio

        )

    )

}

###############################################################
# Epsilon-Greedy Action Selection
###############################################################


epsilon_greedy <- function(

    Q,

    state_id,

    epsilon

){

    q_values <-

        get_Q(

            Q,

            state_id

        )


    ###########################################################
    # Exploration
    ###########################################################

    if(

        runif(1) < epsilon

    ){

        return(

            sample(

                seq_len(N_ACTIONS),

                1

            )

        )

    }


    ###########################################################
    # Greedy action
    ###########################################################

    best <-

        which(

            q_values == max(q_values)

        )


    sample(

        best,

        1

    )

}



###############################################################
# Spatial Action Values
#
# For every possible action a, calculate:
#
#   S_i(a) = sum_j W_ij Q_j(s_j,a)
#
# This is action-specific.
###############################################################


spatial_action_values <- function(

    Q,

    env,

    store_id

){

    action_values <-

        rep(

            0,

            N_ACTIONS

        )


    ###########################################################
    # Identify spatial neighbors
    ###########################################################

    neighbors <-

        which(

            abs(

                W_normalized[store_id, ]

            ) > EPS

        )


    if(

        length(neighbors) == 0

    ){

        return(

            action_values

        )

    }


    ###########################################################
    # Neighbor Q contribution
    ###########################################################

    for(j in neighbors){

        state_j <-

            environment_state_index(

                env,

                j

            )


        q_j <-

            get_Q(

                Q,

                state_j

            )


        action_values <-

            action_values +

            W_normalized[store_id, j] *

            q_j

    }


    action_values

}



###############################################################
# Copula Action Values
#
# For every possible action a:
#
#   C_i(a) = sum_j C_ij Q_j(s_j,a)
#
# This makes copula dependence enter the learning mechanism
# directly rather than only through lambda.
###############################################################


copula_action_values <- function(

    Q,

    env,

    store_id

){

    action_values <-

        rep(

            0,

            N_ACTIONS

        )


    ###########################################################
    # Other stores only
    ###########################################################

    neighbors <-

        which(

            abs(

                Copula_normalized[store_id, ]

            ) > EPS

        )


    neighbors <-

        setdiff(

            neighbors,

            store_id

        )


    if(

        length(neighbors) == 0

    ){

        return(

            action_values

        )

    }


    ###########################################################
    # Copula-weighted neighbor Q values
    ###########################################################

    for(j in neighbors){

        state_j <-

            environment_state_index(

                env,

                j

            )


        q_j <-

            get_Q(

                Q,

                state_j

            )


        action_values <-

            action_values +

            Copula_normalized[store_id, j] *

            q_j

    }


    action_values

}



###############################################################
# Spatial Scalar Value
#
# Used for diagnostics only.
###############################################################


spatial_value <- function(

    Q,

    env,

    store_id

){

    spatial_values <-

        spatial_action_values(

            Q,

            env,

            store_id

        )


    max(

        spatial_values

    )

}



###############################################################
# Copula Scalar Value
###############################################################


copula_value <- function(

    Q,

    env,

    store_id

){

    copula_values <-

        copula_action_values(

            Q,

            env,

            store_id

        )


    max(

}



###############################################################
# Copula Dependence Strength
#
# Unlike the standardized CopulaFeature, this is based directly
# on the magnitude of pairwise dependence.
#
# The diagonal is excluded.
###############################################################


CopulaStrength <-

    numeric(

        N_STORES

    )


for(i in seq_len(N_STORES)){

    other_stores <-

        setdiff(

            seq_len(N_STORES),

            i

        )


    if(

        length(other_stores) > 0

    ){

        CopulaStrength[i] <-

            mean(

                abs(

                    copula_matrix[

                        i,

                        other_stores

                    ]

                ),

                na.rm = TRUE

            )

    }else{

        CopulaStrength[i] <- 0

    }

}



###############################################################
# Protect Copula Strength
###############################################################


CopulaStrength <-

    pmin(

        pmax(

            CopulaStrength,

            0

        ),

        1

    )



###############################################################
# Adaptive LASAR-Q Lambda
#
# Stronger copula dependence -> stronger network coupling.
#
# Unlike the old implementation, this is guaranteed to be
# different from SAR-Q whenever copula dependence is nonzero.
###############################################################


adaptive_lambda <- function(

    store_id

){

    dependence <-

        CopulaStrength[store_id]


    lambda <-

        LASAR_LAMBDA_SPATIAL +

        LASAR_ADAPTIVE_RATE *

        dependence


    lambda <-

        min(

            max(

                lambda,

                0

            ),

            1

        )


    lambda

}



###############################################################
# Adaptive Copula Weight
###############################################################


adaptive_copula_weight <- function(

    store_id

){

    dependence <-

        CopulaStrength[store_id]


    weight <-

        LASAR_LAMBDA_COPULA *

        dependence


    min(

        max(

            weight,

            0

        ),

        1

    )

}



###############################################################
# SAR-Q Action Selection
#
# SAR-Q uses:
#
#   Own Q
#   +
#   spatial neighbor Q
#
# It does NOT use the copula term directly.
###############################################################


select_action_SARQ <- function(

    Q,

    env,

    store_id,

    state_id,

    epsilon

){

    q_values <-

        get_Q(

            Q,

            state_id

        )


    ###########################################################
    # Exploration
    ###########################################################

    if(

        runif(1) < epsilon

    ){

        return(

            sample(

                seq_len(N_ACTIONS),

                1

            )

        )

    }


    ###########################################################
    # Spatial values
    ###########################################################

    spatial_values <-

        spatial_action_values(

            Q,

            env,

            store_id

        )


    ###########################################################
    # SAR-Q score
    ###########################################################

    action_score <-

        (1 - LAMBDA_SPATIAL) *

        q_values +

        LAMBDA_SPATIAL *

        spatial_values


    ###########################################################
    # Greedy action
    ###########################################################

    best <-

        which(

            action_score ==

            max(action_score)

        )


    sample(

        best,

        1

    )

}



###############################################################
# LASAR-Q Action Selection
#
# LASAR-Q uses:
#
#   Own Q
#   +
#   adaptive spatial Q
#   +
#   copula-dependent Q
#
# Therefore LASAR-Q is structurally different from SAR-Q.
###############################################################


select_action_LASARQ <- function(

    Q,

    env,

    store_id,

    state_id,

    epsilon

){

    q_values <-

        get_Q(

            Q,

            state_id

        )


    ###########################################################
    # Exploration
    ###########################################################

    if(

        runif(1) < epsilon

    ){

        return(

            sample(

                seq_len(N_ACTIONS),

                1

            )

        )

    }


    ###########################################################
    # Spatial values
    ###########################################################

    spatial_values <-

        spatial_action_values(

            Q,

            env,

            store_id

        )


    ###########################################################
    # Copula values
    ###########################################################

    copula_values <-

        copula_action_values(

            Q,

            env,

            store_id

        )


    ###########################################################
    # Adaptive spatial weight
    ###########################################################

    lambda_spatial_i <-

        adaptive_lambda(

            store_id

        )


    ###########################################################
    # Adaptive copula weight
    ###########################################################

    lambda_copula_i <-

        adaptive_copula_weight(

            store_id

        )


    ###########################################################
    # Normalize total network weight
    ###########################################################

    network_weight <-

        lambda_spatial_i +

        lambda_copula_i


    if(

        network_weight > 1

    ){

        lambda_spatial_i <-

            lambda_spatial_i /

            network_weight


        lambda_copula_i <-

            lambda_copula_i /

            network_weight

    }


    ###########################################################
    # LASAR-Q action score
    ###########################################################

    own_weight <-

        1 -

        lambda_spatial_i -

        lambda_copula_i


    action_score <-

        own_weight *

        q_values +

        lambda_spatial_i *

        spatial_values +

        lambda_copula_i *

        copula_values


    ###########################################################
    # Greedy action
    ###########################################################

    best <-

        which(

            action_score ==

            max(action_score)

        )


    sample(

        best,

        1

    )

}



###############################################################
# Standard Q-Learning Update
###############################################################


update_Q_learning <- function(

    Q,

    state,

    action,

    reward,

    next_state,

    done = FALSE

){

    q <-

        get_Q(

            Q,

            state

        )


    q_next <-

        get_Q(

            Q,

            next_state

        )


    ###########################################################
    # Terminal-state handling
    ###########################################################

    if(done){

        target <-

            reward

    }else{

        target <-

            reward +

            GAMMA *

            max(

                q_next

            )

    }


    ###########################################################
    # TD update
    ###########################################################

    q[action] <-

        q[action] +

        ALPHA *

        (

            target -

            q[action]

        )


    set_Q(

        Q,

        state,

        q

    )


    Q

}



###############################################################
# SAR-Q Update
#
# SAR-Q:
#
#   target =
#       reward +
#       gamma[
#           (1-lambda) own_value
#           + lambda spatial_value
#       ]
#
# Copula dependence is NOT directly used here.
###############################################################


update_SARQ <- function(

    Q,

    env,

    store_id,

    state,

    action,

    reward,

    next_state,

    done = FALSE

){

    q <-

        get_Q(

            Q,

            state

        )


    ###########################################################
    # Terminal transition
    ###########################################################

    if(done){

        target <-

            reward

    }else{

        #######################################################
        # Own future value
        #######################################################

        q_next <-

            get_Q(

                Q,

                next_state

            )


        own_value <-

            max(

                q_next

            )


        #######################################################
        # Spatial future value
        #######################################################

        neighbor_values <-

            spatial_action_values(

                Q,

                env,

                store_id

            )


        spatial_value_next <-

            max(

                neighbor_values

            )


        #######################################################
        # SAR-Q target
        #######################################################

        target <-

            reward +

            GAMMA *

            (

                (1 - LAMBDA_SPATIAL) *

                own_value +

                LAMBDA_SPATIAL *

                spatial_value_next

            )

    }


    ###########################################################
    # Q update
    #######################################################

    q[action] <-

        q[action] +

        ALPHA *

        (

            target -

            q[action]

        )


    set_Q(

        Q,

        state,

        q

    )


    Q

}



###############################################################
# LASAR-Q Update
#
# LASAR-Q target contains BOTH:
#
#   1. spatial dependence
#   2. copula dependence
#
# with store-specific adaptive weights.
###############################################################


update_LASARQ <- function(

    Q,

    env,

    store_id,

    state,

    action,

    reward,

    next_state,

    done = FALSE

){

    q <-

        get_Q(

            Q,

            state

        )


    ###########################################################
    # Terminal transition
    ###########################################################

    if(done){

        target <-

            reward

    }else{

        #######################################################
        # Own future value
        #######################################################

        q_next <-

            get_Q(

                Q,

                next_state

            )


        own_value <-

            max(

                q_next

            )


        #######################################################
        # Spatial future value
        #######################################################

        spatial_values <-

            spatial_action_values(

                Q,

                env,

                store_id

            )


        spatial_value_next <-

            max(

                spatial_values

            )


        #######################################################
        # Copula future value
        #######################################################

        copula_values <-

            copula_action_values(

                Q,

                env,

                store_id

            )


        copula_value_next <-

            max(

                copula_values

            )


        #######################################################
        # Adaptive weights
        #######################################################

        lambda_spatial_i <-

            adaptive_lambda(

                store_id

            )


        lambda_copula_i <-

            adaptive_copula_weight(

                store_id

            )


        #######################################################
        # Ensure valid convex combination
        #######################################################

        network_weight <-

            lambda_spatial_i +

            lambda_copula_i


        if(

            network_weight > 1

        ){

            lambda_spatial_i <-

                lambda_spatial_i /

                network_weight


            lambda_copula_i <-

                lambda_copula_i /

                network_weight

        }


        own_weight <-

            1 -

            lambda_spatial_i -

            lambda_copula_i


        #######################################################
        # LASAR-Q target
        #######################################################

        target <-

            reward +

            GAMMA *

            (

                own_weight *

                own_value +

                lambda_spatial_i *

                spatial_value_next +

                lambda_copula_i *

                copula_value_next

            )

    }


    ###########################################################
    # Q update
    ###########################################################

    q[action] <-

        q[action] +

        ALPHA *

        (

            target -

            q[action]

        )


    set_Q(

        Q,

        state,

        q

    )


    Q

}



###############################################################
# Method-Specific Action Selection
###############################################################


select_action <- function(

    method,

    Q,

    env,

    store_id,

    state_id,

    epsilon

){

    if(

        method == "Q-Learning"

    ){

        return(

            epsilon_greedy(

                Q,

                state_id,

                epsilon

            )

        )

    }


    if(

        method == "SAR-Q"

    ){

        return(

            select_action_SARQ(

                Q,

                env,

                store_id,

                state_id,

                epsilon

            )

        )

    }


    if(

        method == "LASAR-Q"

    ){

        return(

            select_action_LASARQ(

                Q,

                env,

                store_id,

                state_id,

                epsilon

            )

        )

    }


    stop(

        "Unknown method: ",

        method

    )

}



###############################################################
# Algorithm Selector
###############################################################


RL_update <- function(

    method,

    Q,

    env,

    store_id,

    state,

    action,

    reward,

    next_state,

    done = FALSE

){

    if(

        method == "Q-Learning"

    ){

        return(

            update_Q_learning(

                Q = Q,

                state = state,

                action = action,

                reward = reward,

                next_state = next_state,

                done = done

            )

        )

    }


    if(

        method == "SAR-Q"

    ){

        return(

            update_SARQ(

                Q = Q,

                env = env,

                store_id = store_id,

                state = state,

                action = action,

                reward = reward,

                next_state = next_state,

                done = done

            )

        )

    }


    if(

        method == "LASAR-Q"

    ){

        return(

            update_LASARQ(

                Q = Q,

                env = env,

                store_id = store_id,

                state = state,

                action = action,

                reward = reward,

                next_state = next_state,

                done = done

            )

        )

    }


    stop(

        "Unknown method: ",

        method

    )

}

###############################################################
# SINGLE EPISODE TRAINER
###############################################################


run_episode <- function(

    method,

    Q,

    episode,

    verbose = FALSE

){


    ###########################################################
    # Reset environment
    ###########################################################

    env <-

        reset_environment()



    ###########################################################
    # Total reward by store
    ###########################################################

    total_reward <-

        rep(

            0,

            N_STORES

        )



    ###########################################################
    # Total cost by store
    ###########################################################

    total_cost <-

        rep(

            0,

            N_STORES

        )



    ###########################################################
    # Total shortage by store
    ###########################################################

    total_shortage <-

        rep(

            0,

            N_STORES

        )



    ###########################################################
    # Epsilon schedule
    ###########################################################

    epsilon <-

        max(

            EPSILON_END,

            EPSILON_START *

            EPSILON_DECAY^(episode - 1)

        )



    ###########################################################
    # Episode loop
    ###########################################################

    done <-

        FALSE


    step_count <-

        0L



    while(!done){


        step_count <-

            step_count + 1L



        #######################################################
        # CURRENT STATES
        #######################################################

        states <-

            integer(

                N_STORES

            )



        for(i in seq_len(N_STORES)){

            states[i] <-

                environment_state_index(

                    env,

                    i

                )

        }



        #######################################################
        # ACTION SELECTION
        #######################################################

        actions <-

            integer(

                N_STORES

            )



        for(i in seq_len(N_STORES)){

            actions[i] <-

                select_action(

                    method = method,

                    Q = Q,

                    env = env,

                    store_id = i,

                    state_id = states[i],

                    epsilon = epsilon

                )

        }



        #######################################################
        # ENVIRONMENT TRANSITION
        #######################################################

        transition <-

            environment_step(

                env,

                actions

            )



        #######################################################
        # Next environment
        #######################################################

        new_env <-

            transition$env



        #######################################################
        # Immediate rewards
        #######################################################

        rewards <-

            as.numeric(

                transition$reward

            )



        #######################################################
        # Immediate costs
        #######################################################

        costs <-

            as.numeric(

                transition$cost

            )



        #######################################################
        # Shortage
        #######################################################

        shortage <-

            as.numeric(

                transition$shortage

            )



        #######################################################
        # Terminal indicator
        #######################################################

        done <-

            isTRUE(

                transition$done

            )



        #######################################################
        # NEXT STATES
        #######################################################

        next_states <-

            integer(

                N_STORES

            )



        for(i in seq_len(N_STORES)){

            next_states[i] <-

                environment_state_index(

                    new_env,

                    i

                )

        }



        #######################################################
        # Q-LEARNING / SAR-Q / LASAR-Q UPDATE
        #
        # IMPORTANT:
        #
        # The update receives new_env, not the old env.
        #
        # Therefore spatial/coupla future values are evaluated
        # using the next-period environment.
        #######################################################

        for(i in seq_len(N_STORES)){


            Q <-

                RL_update(

                    method = method,

                    Q = Q,

                    env = new_env,

                    store_id = i,

                    state = states[i],

                    action = actions[i],

                    reward = rewards[i],

                    next_state = next_states[i],

                    done = done

                )

        }



        #######################################################
        # Accumulate performance
        #######################################################

        total_reward <-

            total_reward +

            rewards



        total_cost <-

            total_cost +

            costs



        total_shortage <-

            total_shortage +

            shortage



        #######################################################
        # Optional transition diagnostics
        #######################################################

        if(verbose){

            cat(

                "Episode:",

                episode,

                "Step:",

                step_count,

                "Reward:",

                sum(rewards),

                "Cost:",

                sum(costs),

                "Shortage:",

                sum(shortage),

                "\n"

            )

        }



        #######################################################
        # Move to next environment
        #######################################################

        env <-

            new_env

    }



    ###########################################################
    # FINAL EPISODE KPI
    ###########################################################

    KPI <-

        network_KPI(

            env

        )



    ###########################################################
    # Independent reward/cost consistency check
    ###########################################################

    calculated_reward <-

        -sum(

            total_cost

        )



    observed_reward <-

        sum(

            total_reward

        )



    reward_cost_difference <-

        observed_reward -

        calculated_reward



    ###########################################################
    # Return
    ###########################################################

    list(

        Q = Q,

        reward = total_reward,

        cost = total_cost,

        shortage = total_shortage,

        KPI = KPI,

        epsilon = epsilon,

        steps = step_count,

        reward_cost_difference =

            reward_cost_difference

    )

}



###############################################################
# TRAIN ONE ALGORITHM
###############################################################


train_algorithm <- function(

    method,

    seed

){


    ###########################################################
    # Reproducibility
    ###########################################################

    set.seed(

        seed

    )



    ###########################################################
    # Fresh Q table
    ###########################################################

    Q <-

        create_Q_table()



    ###########################################################
    # Episode reward history
    ###########################################################

    reward_history <-

        numeric(

            EPISODES

        )



    ###########################################################
    # Episode cost history
    ###########################################################

    cost_history <-

        numeric(

            EPISODES

        )



    ###########################################################
    # Episode shortage history
    ###########################################################

    shortage_history <-

        numeric(

            EPISODES

        )



    ###########################################################
    # KPI history
    ###########################################################

    KPI_history <-

        vector(

            "list",

            EPISODES

        )



    ###########################################################
    # Epsilon history
    ###########################################################

    epsilon_history <-

        numeric(

            EPISODES

        )



    ###########################################################
    # Q-table size history
    ###########################################################

    Q_state_count <-

        integer(

            EPISODES

        )



    ###########################################################
    # Reward-cost consistency
    ###########################################################

    reward_cost_difference <-

        numeric(

            EPISODES

        )



    ###########################################################
    # Training loop
    ###########################################################

    for(ep in seq_len(EPISODES)){


        result <-

            run_episode(

                method = method,

                Q = Q,

                episode = ep,

                verbose = FALSE

            )



        #######################################################
        # Update Q table
        #######################################################

        Q <-

            result$Q



        #######################################################
        # Store reward
        #######################################################

        reward_history[ep] <-

            sum(

                result$reward

            )



        #######################################################
        # Store cost
        #######################################################

        cost_history[ep] <-

            sum(

                result$cost

            )



        #######################################################
        # Store shortage
        #######################################################

        shortage_history[ep] <-

            sum(

                result$shortage

            )



        #######################################################
        # Store KPI
        #######################################################

        KPI_history[[ep]] <-

            result$KPI



        #######################################################
        # Epsilon
        #######################################################

        epsilon_history[ep] <-

            result$epsilon



        #######################################################
        # Q-table size
        #######################################################

        Q_state_count[ep] <-

            length(

                ls(

                    envir = Q,

                    all.names = TRUE

                )

            )



        #######################################################
        # Reward-cost consistency
        #######################################################

        reward_cost_difference[ep] <-

            result$reward_cost_difference



        #######################################################
        # Progress
        #######################################################

        if(

            ep %% 50 == 0

        ){

            cat(

                method,

                "Episode",

                ep,

                "| Reward =",

                round(

                    reward_history[ep],

                    3

                ),

                "| Cost =",

                round(

                    cost_history[ep],

                    3

                ),

                "| FillRate =",

                round(

                    KPI_history[[ep]]$FillRate,

                    4

                ),

                "\n"

            )

        }

    }



    ###########################################################
    # Return
    ###########################################################

    list(

        Q = Q,

        reward_history = reward_history,

        cost_history = cost_history,

        shortage_history = shortage_history,

        KPI_history = KPI_history,

        epsilon_history = epsilon_history,

        Q_state_count = Q_state_count,

        reward_cost_difference =

            reward_cost_difference

    )

}

moving_average <-

    function(

        x,

        window = MOVING_WINDOW

    ){

        if(

            length(x) < window

        ){

            return(

                rep(

                    NA_real_,

                    length(x)

                )

            )

        }


        as.numeric(

            stats::filter(

                x,

                rep(

                    1 / window,

                    window

                ),

                sides = 1

            )

        )

    }

###############################################################
# 15. PAIRED MODEL COMPARISON FUNCTION
###############################################################


compare_models <-

    function(

        metric,

        model1,

        model2

    ){


        x <-

            metric %>%

            filter(

                Method == model1

            ) %>%

            select(

                Replication,

                value

            ) %>%

            arrange(

                Replication

            )



        y <-

            metric %>%

            filter(

                Method == model2

            ) %>%

            select(

                Replication,

                value

            ) %>%

            arrange(

                Replication

            )



        #######################################################
        # Check sample sizes
        #######################################################

        if(

            nrow(x) !=

            nrow(y)

        ){

            stop(

                paste0(

                    "Unequal numbers of replications: ",

                    model1,

                    " = ",

                    nrow(x),

                    ", ",

                    model2,

                    " = ",

                    nrow(y)

                )

            )

        }



        #######################################################
        # Check replication matching
        #######################################################

        if(

            !identical(

                x$Replication,

                y$Replication

            )

        ){

            stop(

                paste0(

                    "Replication IDs do not match between ",

                    model1,

                    " and ",

                    model2,

                    "."

                )

            )

        }



        #######################################################
        # Paired difference
        #######################################################

        differences <-

            x$value -

            y$value



        #######################################################
        # Wilcoxon signed-rank test
        #######################################################

        wilcox <-

            wilcox.test(

                x$value,

                y$value,

                paired = TRUE,

                exact = FALSE

            )



        #######################################################
        # Paired t-test
        #
        # Included as a secondary parametric analysis.
        #######################################################

        paired_t <-

            t.test(

                x$value,

                y$value,

                paired = TRUE

            )



        #######################################################
        # Paired standardized effect
        #
        # Cohen's dz
        #######################################################

        dz <-

            mean(

                differences,

                na.rm = TRUE

            ) /

            sd(

                differences,

                na.rm = TRUE

            )



        #######################################################
        # Cliff's delta
        #
        # Computed as a supplementary effect-size measure.
        #######################################################

        cliff <-

            effsize::cliff.delta(

                x$value,

                y$value

            )



        #######################################################
        # Return
        #######################################################

        data.frame(

            Model1 =

                model1,

            Model2 =

                model2,

            N =

                length(

                    differences

                ),

            MeanDifference =

                mean(

                    differences,

                    na.rm = TRUE

                ),

            MedianDifference =

                median(

                    differences,

                    na.rm = TRUE

                ),

            SDDifference =

                sd(

                    differences,

                    na.rm = TRUE

                ),

            PairedT =

                as.numeric(

                    paired_t$statistic

                ),

            PairedTPValue =

                paired_t$p.value,

            WilcoxonV =

                as.numeric(

                    wilcox$statistic

                ),

            WilcoxonPValue =

                wilcox$p.value,

            Cohen_dz =

                dz,

            CliffsDelta =

                as.numeric(

                    cliff$estimate

                ),

            stringsAsFactors = FALSE

        )

}

# 21. BOOTSTRAP FUNCTION
###############################################################


bootstrap_difference <-

    function(

        data,

        indices

    ){

        d <-

            data$Difference[

                indices

            ]


        mean(

            d,

            na.rm = TRUE

        )

    }



###############################################################
# 22. BOOTSTRAP MODEL COMPARISON
###############################################################


bootstrap_model_difference <-

    function(

        metric,

        model1,

        model2,

        R = BOOT_R

    ){


        x <-

            metric %>%

            filter(

                Method == model1

            ) %>%

            select(

                Replication,

                value

            ) %>%

            arrange(

                Replication

            )



        y <-

            metric %>%

            filter(

                Method == model2

            ) %>%

            select(

                Replication,

                value

            ) %>%

            arrange(

                Replication

            )



        if(

            !identical(

                x$Replication,

                y$Replication

            )

        ){

            stop(

                paste0(

                    "Replication IDs do not match between ",

                    model1,

                    " and ",

                    model2

                )

            )

        }



        difference_data <-

            data.frame(

                Replication =

                    x$Replication,

                Difference =

                    x$value -

                    y$value

            )



        #######################################################
        # Observed difference
        #######################################################

        observed <-

            mean(

                difference_data$Difference,

                na.rm = TRUE

            )



        #######################################################
        # Bootstrap
        #######################################################

        boot_result <-

            boot(

                data = difference_data,

                statistic = bootstrap_difference,

                R = R

            )



        #######################################################
        # Percentile CI
        #######################################################

        boot_ci <-

            boot.ci(

                boot_result,

                type = "perc"

            )



        #######################################################
        # Return
        #######################################################

        data.frame(

            Model1 =

                model1,

            Model2 =

                model2,

            N =

                nrow(

                    difference_data

                ),

            ObservedMeanDifference =

                observed,

            BootstrapMeanDifference =

                mean(

                    boot_result$t,

                    na.rm = TRUE

                ),

            CI_Lower =

                if(

                    !is.null(

                        boot_ci$percent

                    )

                )

                boot_ci$percent[4]

                else

                    NA_real_,

            CI_Upper =

                if(

                    !is.null(

                        boot_ci$percent

                    )

                )

                boot_ci$percent[5]

                else

                    NA_real_

        )

    }

###############################################################
# 7. SCENARIO DEMAND GENERATOR
###############################################################

generate_scenario_demand <- function(

    scenario,

    day,

    demand_multiplier = 1

) {


    base <-

        as.numeric(

            sales_matrix[day, ]

        )


    ###########################################################
    # NORMAL
    ###########################################################

    if (
        scenario == "Normal"
    ) {

        return(

            pmax(

                round(

                    base *

                    demand_multiplier

                ),

                0

            )

        )

    }


    ###########################################################
    # HIGH VOLATILITY
    ###########################################################

    if (
        scenario == "High_Volatility"
    ) {

        volatility_factor <-

            rlnorm(

                N_STORES,

                meanlog = 0,

                sdlog = 0.30

            )


        return(

            pmax(

                round(

                    base *

                    volatility_factor *

                    demand_multiplier

                ),

                0

            )

        )

    }


    ###########################################################
    # DEMAND SHOCK
    ###########################################################

    if (
        scenario == "Demand_Shock"
    ) {

        shock <-

            ifelse(

                day %% 30 == 0,

                2,

                1

            )


        return(

            pmax(

                round(

                    base *

                    shock *

                    demand_multiplier

                ),

                0

            )

        )

    }


    ###########################################################
    # OTHER SCENARIOS
    ###########################################################

    pmax(

        round(

            base *

            demand_multiplier

        ),

        0

    )

}

apply_scenario_parameters <- function(

    env,

    scenario_settings

) {


    ###########################################################
    # Lead time
    ###########################################################

    if (
        "LEAD_TIME" %in%
        names(env)
    ) {

        env$LEAD_TIME <-
            scenario_settings$LeadTime

    }


    if (
        "lead_time" %in%
        names(env)
    ) {

        env$lead_time <-
            scenario_settings$LeadTime

    }


    ###########################################################
    # Service level
    ###########################################################

    if (
        "SERVICE_LEVEL" %in%
        names(env)
    ) {

        env$SERVICE_LEVEL <-
            scenario_settings$ServiceTarget

    }


    if (
        "service_level" %in%
        names(env)
    ) {

        env$service_level <-
            scenario_settings$ServiceTarget

    }


    env

}

###############################################################
# 10. ROBUSTNESS EXPERIMENT
#
# IMPORTANT:
#
# The experiment now:
#
# 1. Uses the correct method-specific policy.
# 2. Allows a trained Q table to be supplied.
# 3. Uses the scenario demand before state construction.
# 4. Calculates final KPI using network_KPI().
###############################################################

run_robustness_experiment <- function(

    scenario_name,

    method,

    replication = 1,

    scenario_settings = NULL,

    trained_Q = NULL,

    epsilon = 0.05

) {


    ###########################################################
    # Scenario settings
    ###########################################################

    if (
        is.null(
            scenario_settings
        )
    ) {

        scenario_settings <-

            scenarios %>%

            filter(
                Scenario == scenario_name
            )

    }


    if (
        nrow(
            scenario_settings
        ) != 1
    ) {

        stop(

            paste(
                "Scenario not found:",
                scenario_name
            )

        )

    }


    ###########################################################
    # Reproducible replication
    ###########################################################

    set.seed(

        10000 +

        replication

    )


    ###########################################################
    # Reset environment
    ###########################################################

    env <-

        reset_environment()


    ###########################################################
    # Apply scenario parameters
    ###########################################################

    env <-

        apply_scenario_parameters(

            env,

            scenario_settings

        )


    ###########################################################
    # Q table
    #
    # If a trained Q table is supplied, use it.
    # Otherwise create a new table.
    ###########################################################

    if (
        is.null(
            trained_Q
        )
    ) {

        Q <-

            create_Q_table()

    } else {

        Q <-

            trained_Q

    }


    ###########################################################
    # Storage
    ###########################################################

    rewards <-

        numeric(
            N_DAYS
        )


    ###########################################################
    # Simulation
    ###########################################################

    for (

        day in seq_len(
            N_DAYS
        )

    ) {


        #######################################################
        # Set current scenario demand BEFORE state construction
        #######################################################

        env$demand <-

            generate_scenario_demand(

                scenario =
                    scenario_name,

                day =
                    day,

                demand_multiplier =
                    scenario_settings$DemandMultiplier

            )


        #######################################################
        # States
        #######################################################

        states <-

            integer(
                N_STORES
            )


        #######################################################
        # Actions
        #######################################################

        actions <-

            integer(
                N_STORES
            )


        #######################################################
        # Action selection
        #######################################################

        for (
            i in seq_len(
                N_STORES
            )
        ) {


            states[i] <-

                environment_state_index(

                    env,

                    i

                )


            actions[i] <-

                select_robustness_action(

                    method =
                        method,

                    Q =
                        Q,

                    env =
                        env,

                    store_id =
                        i,

                    state_id =
                        states[i],

                    epsilon =
                        epsilon

                )

        }


        #######################################################
        # Environment transition
        #######################################################

        transition <-

            environment_step(

                env,

                actions

            )


        new_env <-

            transition$env


        rewards_day <-

            transition$reward


        #######################################################
        # Store reward
        #######################################################

        rewards[day] <-

            sum(

                rewards_day,

                na.rm = TRUE

            )


        #######################################################
        # Update Q-values if training mode is requested
        #
        # For a trained-policy robustness evaluation,
        # Q is kept fixed.
        #######################################################

        if (
            is.null(
                trained_Q
            )
        ) {

            next_states <-

                integer(
                    N_STORES
                )


            for (
                i in seq_len(
                    N_STORES
                )
            ) {


                next_states[i] <-

                    environment_state_index(

                        new_env,

                        i

                    )


                Q <-

                    RL_update(

                        method,

                        Q,

                        new_env,

                        i,

                        states[i],

                        actions[i],

                        rewards_day[i],

                        next_states[i]

                    )

            }

        }


        #######################################################
        # Advance environment
        #######################################################

        env <-

            new_env

    }


    ###########################################################
    # Final KPI
    ###########################################################

    final_KPI <-

        network_KPI(

            env

        )


    ###########################################################
    # Extract KPI safely
    ###########################################################

    get_kpi_value <- function(

        x,

        name,

        default = NA_real_

    ) {

        if (
            name %in%
            names(x)
        ) {

            return(
                as.numeric(
                    x[[name]]
                )
            )

        }

        default

    }


    total_reward <-

        sum(

            rewards,

            na.rm = TRUE

        )


    total_cost <-

        get_kpi_value(

            final_KPI,

            "TotalCost"

        )


    fill_rate <-

        get_kpi_value(

            final_KPI,

            "FillRate"

        )


    stockouts <-

        get_kpi_value(

            final_KPI,

            "TotalStockouts"

        )


    ###########################################################
    # Return
    ###########################################################

    data.frame(

        Replication =
            replication,

        Scenario =
            scenario_name,

        Method =
            method,

        TotalReward =
            total_reward,

        MeanReward =
            mean(
                rewards,
                na.rm = TRUE
            ),

        SDReward =
            sd(
                rewards,
                na.rm = TRUE
            ),

        TotalCost =
            total_cost,

        FillRate =
            fill_rate,

        TotalStockouts =
            stockouts,

        stringsAsFactors =
            FALSE

    )

}

generate_copula_feature <- function(

    copula_name

) {


    if (
        copula_name == "Gaussian"
    ) {

        return(

            safe_scale(

                rowMeans(

                    copula_matrix,

                    na.rm = TRUE

                )

            )

        )

    }


    if (
        copula_name == "Clayton"
    ) {

        return(

            safe_scale(

                rowMeans(

                    copula_matrix^2,

                    na.rm = TRUE

                )

            )

        )

    }


    if (
        copula_name == "Gumbel"
    ) {

        return(

            safe_scale(

                rowMeans(

                    -log(

                        pmax(

                            copula_matrix,

                            1e-8

                        )

                    ),

                    na.rm = TRUE

                )

            )

        )

    }


    stop(

        paste(
            "Unknown copula:",
            copula_name
        )

    )

}

###############################################################
# PART 4
#
# Base-Stock Policy
#
###############################################################

BASE_SERVICE_LEVEL <-
    0.95


base_stock_policy <- function(
    env
) {

    ###########################################################
    # Safety-stock component
    ###########################################################

    safety_stock <-
        qnorm(
            BASE_SERVICE_LEVEL
        ) *
        demand_sd *
        sqrt(
            LEAD_TIME
        )


    ###########################################################
    # Target inventory
    ###########################################################

    target_inventory <-
        demand_base *
        LEAD_TIME +
        safety_stock


    ###########################################################
    # Order quantity
    ###########################################################

    order_qty <-
        pmax(
            target_inventory -
            env$inventory,
            0
        )


    ###########################################################
    # Convert order quantity to action
    ###########################################################

    action <-
        findInterval(
            order_qty,
            ACTION_SPACE
        )


    ###########################################################
    # Protect action boundaries
    ###########################################################

    action <-
        pmin(
            pmax(
                action,
                1
            ),
            N_ACTIONS
        )


    return(
        action
    )

}


###############################################################
# PART 5
#
# Environment Output Helper
#
###############################################################

get_result_vector <- function(
    result,
    env,
    variable,
    default = 0
) {

    ###########################################################
    # Prefer variable returned directly by environment_step
    ###########################################################

    if (
        variable %in%
        names(result)
    ) {

        value <-
            result[[variable]]

        if (
            length(value) > 0
        ) {

            return(
                value
            )

        }

    }


    ###########################################################
    # Otherwise use updated environment
    ###########################################################

    if (
        variable %in%
        names(env)
    ) {

        value <-
            env[[variable]]

        if (
            length(value) > 0
        ) {

            return(
                value
            )

        }

    }


    ###########################################################
    # Default
    ###########################################################

    return(
        default
    )

}


###############################################################
# PART 6
#
# Evaluate Policy
#
###############################################################

evaluate_policy <- function(
    policy,
    method_name,
    replication
) {

    ###########################################################
    # Reset environment
    ###########################################################

    env <-
        reset_environment()


    ###########################################################
    # Performance storage
    ###########################################################

    total_reward <-
        0

    total_cost <-
        0

    total_stockout <-
        0

    total_sales <-
        0

    total_demand <-
        0


    ###########################################################
    # Daily simulation
    ###########################################################

    for (
        day in seq_len(
            N_DAYS
        )
    ) {

        #######################################################
        # Select action
        #######################################################

        actions <-
            policy(
                env
            )


        #######################################################
        # Validate actions
        #######################################################

        if (
            length(actions) !=
            N_STORES
        ) {

            stop(
                paste(
                    "Invalid action length for method:",
                    method_name
                )
            )

        }


        #######################################################
        # Environment step
        #######################################################

        result <-
            environment_step(
                env,
                actions
            )


        #######################################################
        # Update environment
        #######################################################

        env <-
            result$env


        #######################################################
        # Reward
        #######################################################

        reward_value <-
            get_result_vector(
                result,
                env,
                "reward"
            )

        total_reward <-
            total_reward +
            sum(
                reward_value,
                na.rm = TRUE
            )


        #######################################################
        # Cost
        #######################################################

        cost_value <-
            get_result_vector(
                result,
                env,
                "cost"
            )

        total_cost <-
            total_cost +
            sum(
                cost_value,
                na.rm = TRUE
            )


        #######################################################
        # Stockouts
        #######################################################

        stockout_value <-
            get_result_vector(
                result,
                env,
                "stockout"
            )

        total_stockout <-
            total_stockout +
            sum(
                stockout_value,
                na.rm = TRUE
            )


        #######################################################
        # Sales
        #######################################################

        sales_value <-
            get_result_vector(
                result,
                env,
                "sales"
            )

        total_sales <-
            total_sales +
            sum(
                sales_value,
                na.rm = TRUE
            )


        #######################################################
        # Demand
        #######################################################

        demand_value <-
            get_result_vector(
                result,
                env,
                "demand"
            )

        total_demand <-
            total_demand +
            sum(
                demand_value,
                na.rm = TRUE
            )

    }


    ###########################################################
    # Fill rate
    ###########################################################

    fill_rate <-
        total_sales /
        pmax(
            total_demand,
            1
        )


    ###########################################################
    # Return replication-level result
    ###########################################################

    data.frame(

        Replication =
            replication,

        Method =
            method_name,

        TotalReward =
            total_reward,

        TotalCost =
            total_cost,

        FillRate =
            fill_rate,

        Stockouts =
            total_stockout,

        TotalSales =
            total_sales,

        TotalDemand =
            total_demand,

        stringsAsFactors =
            FALSE
    )

}


###############################################################
# PART 7
#
# RL Policy Extraction
#
###############################################################

create_RL_policy <- function(
    Q,
    method
) {

    ###########################################################
    # Check Q table
    ###########################################################

    if (
        is.null(Q)
    ) {

        stop(
            paste(
                "Q table is NULL for method:",
                method
            )
        )

    }


    ###########################################################
    # Create deterministic policy
    ###########################################################

    policy_function <-
        function(
            env
        ) {

            actions <-
                integer(
                    N_STORES
                )


            ###################################################
            # Select action for each store
            ###################################################

            for (
                i in seq_len(
                    N_STORES
                )
            ) {

                state <-
                    environment_state_index(
                        env,
                        i
                    )


                q_values <-
                    get_Q(
                        Q,
                        state
                    )


                if (
                    length(q_values) == 0
                ) {

                    stop(
                        paste(
                            "No Q-values found for state:",
                            state
                        )
                    )

                }


                ################################################
                # Greedy action
                ################################################

                actions[i] <-
                    which.max(
                        q_values
                    )

            }


            return(
                actions
            )

        }


    return(
        policy_function
    )

}

benchmark_test <- function(

    metric,

    m1,

    m2

) {

    ###########################################################
    # Extract data
    ###########################################################

    x <-
        benchmark_results %>%

        filter(
            Method == m1
        ) %>%

        select(
            Replication,
            value = all_of(metric)
        )


    y <-
        benchmark_results %>%

        filter(
            Method == m2
        ) %>%

        select(
            Replication,
            value = all_of(metric)
        )


    ###########################################################
    # Match replications
    ###########################################################

    paired_data <-
        inner_join(

            x,

            y,

            by =
                "Replication",

            suffix =
                c(
                    "_1",
                    "_2"
                )
        )


    ###########################################################
    # Check sample size
    ###########################################################

    if (
        nrow(
            paired_data
        ) < 2
    ) {

        return(
            data.frame(

                Model1 =
                    m1,

                Model2 =
                    m2,

                Metric =
                    metric,

                N =
                    nrow(
                        paired_data
                    ),

                PValue =
                    NA_real_,

                CliffDelta =
                    NA_real_

            )
        )

    }


    ###########################################################
    # Paired vectors
    ###########################################################

    x_value <-
        paired_data$value_1

    y_value <-
        paired_data$value_2


    ###########################################################
    # Paired Wilcoxon test
    ###########################################################

    wilcox <-
        wilcox.test(

            x_value,

            y_value,

            paired =
                TRUE,

            exact =
                FALSE
        )


    ###########################################################
    # Paired Cliff's delta
    ###########################################################

    effect <-
        effsize::cliff.delta(

            x_value,

            y_value,

            paired =
                TRUE
        )


    ###########################################################
    # Return
    ###########################################################

    data.frame(

        Model1 =
            m1,

        Model2 =
            m2,

        Metric =
            metric,

        N =
            nrow(
                paired_data
            ),

        PValue =
            wilcox$p.value,

        CliffDelta =
            as.numeric(
                effect$estimate
            ),

        stringsAsFactors =
            FALSE
    )

}

flatten_column <- function(x) {

    if (is.list(x)) {

        x <- unlist(
            x,
            recursive = TRUE,
            use.names = FALSE
        )

    }

    return(x)

}


###############################################################
# Normalize benchmark data columns
###############################################################

normalize_benchmark_columns <- function(df) {

    columns_to_normalize <- c(
        "Method",
        "Replication",
        "TotalReward",
        "Reward",
        "TotalCost",
        "Cost",
        "FillRate",
        "Stockouts"
    )

    for (nm in columns_to_normalize) {

        if (nm %in% names(df)) {

            df[[nm]] <-
                flatten_column(
                    df[[nm]]
                )

        }

    }

    ###########################################################
    # Standardize Method
    ###########################################################

    if ("Method" %in% names(df)) {

        df$Method <-
            as.character(
                df$Method
            )

    }

    ###########################################################
    # Standardize Replication
    ###########################################################

    if ("Replication" %in% names(df)) {

        df$Replication <-
            suppressWarnings(
                as.integer(
                    df$Replication
                )
            )

    }

    ###########################################################
    # Standardize numeric variables
    ###########################################################

    numeric_columns <- c(
        "TotalReward",
        "Reward",
        "TotalCost",
        "Cost",
        "FillRate",
        "Stockouts"
    )

    for (nm in numeric_columns) {

        if (nm %in% names(df)) {

            df[[nm]] <-
                suppressWarnings(
                    as.numeric(
                        df[[nm]]
                    )
                )

        }

    }

    return(df)

}


###############################################################
# Safe percentage calculation
###############################################################

safe_percent_change <- function(
    old_value,
    new_value
) {

    if (
        length(old_value) != 1 ||
        length(new_value) != 1 ||
        is.na(old_value) ||
        is.na(new_value) ||
        !is.finite(old_value) ||
        !is.finite(new_value) ||
        abs(old_value) < .Machine$double.eps
    ) {

        return(
            NA_real_
        )

    }

    return(
        100 *
        (
            old_value -
            new_value
        ) /
        old_value
    )

}


###############################################################
# Safe reward improvement
###############################################################

safe_reward_improvement <- function(
    baseline,
    new_value
) {

    if (
        length(baseline) != 1 ||
        length(new_value) != 1 ||
        is.na(baseline) ||
        is.na(new_value) ||
        !is.finite(baseline) ||
        !is.finite(new_value) ||
        abs(baseline) < .Machine$double.eps
    ) {

        return(
            NA_real_
        )

    }

    return(
        100 *
        (
            new_value -
            baseline
        ) /
        abs(baseline)
    )

}


###############################################################
# Pareto efficiency function
###############################################################

is_pareto_efficient <- function(
    cost,
    fill_rate
) {

    n <-
        length(cost)

    efficient <-
        rep(
            FALSE,
            n
        )

    valid <-
        is.finite(cost) &
        is.finite(fill_rate)

    if (!any(valid)) {

        return(
            efficient
        )

    }

    valid_indices <-
        which(valid)

    for (i in valid_indices) {

        dominated <-
            FALSE

        for (j in valid_indices) {

            if (i == j) {
                next
            }

            if (
                cost[j] <= cost[i] &&
                fill_rate[j] >= fill_rate[i] &&
                (
                    cost[j] < cost[i] ||
                    fill_rate[j] > fill_rate[i]
                )
            ) {

                dominated <-
                    TRUE

                break

            }

        }

        efficient[i] <-
            !dominated

    }

    return(
        efficient
    )

}

# PART 10

# Spatial Network
#
###############################################################

if (
    !is.null(
        W_matrix
    )
) {

    ###########################################################
    # Convert to numeric matrix
    ###########################################################

    W_matrix <-
        as.matrix(
            W_matrix
        )


    storage.mode(
        W_matrix
    ) <-
        "numeric"


    ###########################################################
    # Remove diagonal
    ###########################################################

    diag(
        W_matrix
    ) <-
        0


    ###########################################################
    # Symmetrize
    ###########################################################

    W_network <-
        (
            W_matrix +
            t(W_matrix)
        ) / 2


    ###########################################################
    # Replace missing values
    ###########################################################

    W_network[
        !is.finite(
            W_network
        )
    ] <-
        0


    ###########################################################
    # Create graph
    ###########################################################

    network_graph <-
        igraph::graph_from_adjacency_matrix(

            W_network,

            mode =
                "undirected",

            weighted =
                TRUE,

            diag =
                FALSE

        )


    ###########################################################
    # Store labels
    ###########################################################

    if (

        !is.null(
            selected_stores
        ) &&

        length(
            selected_stores
        ) ==
        igraph::vcount(
            network_graph
        )

    ) {

        V(
            network_graph
        )$label <-
            as.character(
                selected_stores
            )

    } else {

        V(
            network_graph
        )$label <-
            as.character(
                seq_len(
                    igraph::vcount(
                        network_graph
                    )
                )
            )

    }


    ###########################################################
    # Edge width
    ###########################################################

    edge_weights <-
        E(
            network_graph
        )$weight


    if (
        length(edge_weights) > 0
    ) {

        edge_width <-
            1 +
            4 *
            pmax(
                0,
                abs(
                    edge_weights
                )
            )

    } else {

        edge_width <-
            numeric(0)

    }


    ###########################################################
    # Network plot
    ###########################################################

    png(

        "results/Figure5_SpatialNetwork.png",

        width = 1000,

        height = 800,

        res = 120

    )


    plot(

        network_graph,

        vertex.label =
            V(network_graph)$label,

        vertex.size =
            28,

        vertex.color =
            "lightgray",

        vertex.frame.color =
            "black",

        edge.width =
            edge_width,

        edge.arrow.size =
            0,

        main =
            "Spatial Store Dependence Network"

    )


    dev.off()

}

flatten_column <- function(x) {

    if (is.list(x)) {

        x <-
            unlist(
                x,
                recursive = TRUE,
                use.names = FALSE
            )

    }

    return(x)

}


normalize_dataframe <- function(df) {

    if ("Method" %in% names(df)) {

        df$Method <-
            as.character(
                flatten_column(
                    df$Method
                )
            )

    }

    return(df)

}


safe_read_csv <- function(
    filename,
    required = TRUE
) {

    path <-
        file.path(
            RESULT_DIR,
            filename
        )

    if (!file.exists(path)) {

        if (required) {

            stop(
                paste(
                    "Required result file not found:",
                    path
                )
            )

        } else {

            warning(
                paste(
                    "Optional result file not found:",
                    path
                )
            )

            return(NULL)

        }

    }

    df <-
        read.csv(
            path,
            stringsAsFactors = FALSE
        )

    return(
        normalize_dataframe(df)
    )

}
