###############################################################################
# 2. DYNAMIC LONGITUDINAL SIMULATION ENGINE (SEQUENTIAL IGNORABILITY)
###############################################################################
# Pre-treatment covariates X_{i, t-1} and treatment history A_{i, t-1} plus
# known game calendar at time t determine treatment assignment A_{it} and
# potential outcomes Y_{it}(0), Y_{it}(1). Post-treatment state X_{it} evolves
# *after* A_{it} is realized, ensuring zero unobserved contemporaneous confounding.

simulate_dynamic_sports_data <- function(n = N_CUSTOMERS, t_steps = T_TIMESTEPS,
                                         p = P_COVARIATES, rho = 0.65) {
  X_panel    <- array(0, dim = c(n, t_steps, p))
  CAL_panel  <- matrix(0, nrow = n, ncol = t_steps) # Known game-day schedule
  A_panel    <- matrix(0, nrow = n, ncol = t_steps)
  Y_panel    <- matrix(0, nrow = n, ncol = t_steps)
  TAU_panel  <- matrix(0, nrow = n, ncol = t_steps)
  PROP_panel <- matrix(0, nrow = n, ncol = t_steps)
  Y0_panel   <- matrix(0, nrow = n, ncol = t_steps)
  Y1_panel   <- matrix(0, nrow = n, ncol = t_steps)

  for (i in 1:n) {
    alpha_i <- rnorm(p, mean = 0, sd = 0.5)
    X_panel[i, 1, ] <- alpha_i + rnorm(p, mean = 0, sd = 0.8)

    for (t in 1:t_steps) {
      # Known upcoming game-calendar index (weekly match cycle + season arc)
      cal_t <- sin(2 * pi * t / 7) + 0.5 * cos(2 * pi * t / 30)
      CAL_panel[i, t] <- cal_t

      # Observable pre-decision state (from t-1)
      X_prev <- if (t == 1) X_panel[i, 1, ] else X_panel[i, t - 1, ]
      A_prev <- if (t == 1) 0 else A_panel[i, t - 1]
      A_lag2 <- if (t <= 2) 0 else A_panel[i, t - 2]

      # Dynamic Propensity Score e(H_it) based strictly on observable history
      logit_e <- -0.35 + 0.35 * X_prev[1] - 0.30 * X_prev[2] +
        0.25 * X_prev[3] + 0.20 * cal_t - 0.45 * A_prev
      prop_it <- pmin(pmax(plogis(logit_e), 0.05), 0.95)
      PROP_panel[i, t] <- prop_it

      # Historical treatment assignment
      A_it <- rbinom(1, 1, prop_it)
      A_panel[i, t] <- A_it

      # Non-linear Dynamic CATE tau(H_it): captures temporal momentum,
      # game schedule synergy, and promotional fatigue from repeated targeting
      momentum_1 <- if (t >= 3) (X_panel[i, t - 1, 1] - X_panel[i, t - 3, 1]) else 0
      fatigue <- 0.35 * A_prev + 0.15 * A_lag2

      tau_it <- 0.42 +
        0.35 * sin(X_prev[1]) +
        0.30 * momentum_1 +
        0.25 * X_prev[2] * X_prev[3] +
        0.20 * (X_prev[4]^2 - 1) +
        0.18 * cal_t -
        fatigue
      TAU_panel[i, t] <- tau_it

      # Baseline Ticket Revenue Y(0)
      y0_it <- 1.00 + 0.25 * X_prev[1] - 0.10 * X_prev[2] + 0.20 * cal_t
      Y0_panel[i, t] <- y0_it
      Y1_panel[i, t] <- y0_it + tau_it

      # Realized Outcome with observation noise
      eps <- rnorm(1, mean = 0, sd = 0.35)
      Y_panel[i, t] <- ifelse(A_it == 1, Y1_panel[i, t] + eps, Y0_panel[i, t] + eps)

      # Post-decision state evolution X_{it} (realized after A_{it})
      if (t > 1) {
        X_panel[i, t, ] <- rho * X_panel[i, t - 1, ] +
          (1 - rho) * alpha_i +
          0.12 * cal_t +
          0.15 * A_it +
          rnorm(p, mean = 0, sd = 0.3)
      }
    }
  }

  return(list(
    X = X_panel, CAL = CAL_panel, A = A_panel, Y = Y_panel,
    TAU = TAU_panel, PROP = PROP_panel, Y0 = Y0_panel, Y1 = Y1_panel
  ))
}

###############################################################################
# 3. CONSTRUCT TEMPORAL LOOKBACK SEQUENCES (MATCHED INFORMATION SETS)
###############################################################################
# Each sequence step k in 1:lookback aligns pre-decision covariates X_{t-L+k-1},
# prior treatment A_{t-L+k-1}, and the target-period game calendar CAL_t,
# ensuring the model has the exact same information set H_it as the DGP.

build_chronological_dataset <- function(sim_data, lookback = LOOKBACK) {
  X_arr   <- sim_data$X
  CAL_mat <- sim_data$CAL
  A_mat   <- sim_data$A
  Y_mat   <- sim_data$Y
  TAU_mat <- sim_data$TAU
  PROP_mat <- sim_data$PROP
  Y0_mat  <- sim_data$Y0
  Y1_mat  <- sim_data$Y1

  n       <- dim(X_arr)[1]
  t_steps <- dim(X_arr)[2]
  p       <- dim(X_arr)[3]

  # Total input channels per timestep = p covariates + prior A + target calendar
  p_total <- p + 2
  seq_per_fan <- t_steps - lookback
  total_seq   <- n * seq_per_fan

  CUST_id  <- integer(total_seq)
  X_seq    <- array(0, dim = c(total_seq, lookback, p_total))
  FLAT_seq <- matrix(0, nrow = total_seq, ncol = lookback * p_total)
  RFM_seq  <- matrix(0, nrow = total_seq, ncol = p_total * 2)
  A_seq    <- numeric(total_seq)
  Y_seq    <- numeric(total_seq)
  TAU_seq  <- numeric(total_seq)
  PROP_seq <- numeric(total_seq)
  Y0_seq   <- numeric(total_seq)
  Y1_seq   <- numeric(total_seq)

  idx <- 1
  for (i in 1:n) {
    for (t in (lookback + 1):t_steps) {
      hist_range <- (t - lookback):(t - 1)
      # Assemble full pre-decision history matrix (lookback x (p + 2))
      seq_slice <- cbind(
        X_arr[i, hist_range, , drop = FALSE][1, , ],
        A_mat[i, hist_range],
        rep(CAL_mat[i, t], lookback)
      )

      CUST_id[idx]    <- i
      X_seq[idx, , ]  <- seq_slice
      FLAT_seq[idx, ] <- as.vector(t(seq_slice)) # Full flattened history for fair ML baselines
      RFM_seq[idx, ]  <- c(colMeans(seq_slice), apply(seq_slice, 2, sd))

      A_seq[idx]    <- A_mat[i, t]
      Y_seq[idx]    <- Y_mat[i, t]
      TAU_seq[idx]  <- TAU_mat[i, t]
      PROP_seq[idx] <- PROP_mat[i, t]
      Y0_seq[idx]   <- Y0_mat[i, t]
      Y1_seq[idx]   <- Y1_mat[i, t]

      idx <- idx + 1
    }
  }

  return(list(
    CUST_id = CUST_id, X_seq = X_seq, FLAT_seq = FLAT_seq, RFM_seq = RFM_seq,
    A_seq = A_seq, Y_seq = Y_seq, TAU_seq = TAU_seq, PROP_seq = PROP_seq,
    Y0_seq = Y0_seq, Y1_seq = Y1_seq
  ))
}

###############################################################################
# 4. TEMPORAL CAUSAL DEEP ARCHITECTURE & ABLATION VARIANTS
###############################################################################
# Supports:
#   - "full"        : CNN + BiLSTM + Multi-Head Attention + Propensity Head
#   - "cnn"         : 1D-CNN only + Propensity Head
#   - "bilstm"      : BiLSTM only + Propensity Head
#   - "transformer" : Multi-Head Attention only + Propensity Head
#   - "tarnet"      : Full 3-branch encoder WITHOUT Propensity Head (loss_weight=0)

build_temporal_causal_network <- function(lookback = LOOKBACK, n_features = P_COVARIATES + 2,
                                          branch_type = "full") {
  inputs <- layer_input(shape = c(lookback, n_features), name = "temporal_history")

  if (branch_type %in% c("full", "tarnet", "cnn")) {
    cnn_branch <- inputs %>%
      layer_conv_1d(filters = 32, kernel_size = 3, padding = "same", activation = "relu") %>%
      layer_batch_normalization() %>%
      layer_global_average_pooling_1d()
  }

  if (branch_type %in% c("full", "tarnet", "bilstm")) {
    bilstm_branch <- keras_layers$Bidirectional(
      keras_layers$LSTM(units = 32L, return_sequences = FALSE)
    )(inputs)
  }

  if (branch_type %in% c("full", "tarnet", "transformer")) {
    attn <- layer_multi_head_attention(num_heads = 2, key_dim = 16)(
      query = inputs, key = inputs, value = inputs
    )
    attn_add <- layer_add(list(inputs, attn)) %>% layer_layer_normalization()
    transformer_branch <- attn_add %>% layer_global_average_pooling_1d()
  }

  combined <- switch(
    branch_type,
    "full"        = layer_concatenate(list(cnn_branch, bilstm_branch, transformer_branch)),
    "tarnet"      = layer_concatenate(list(cnn_branch, bilstm_branch, transformer_branch)),
    "cnn"         = cnn_branch,
    "bilstm"      = bilstm_branch,
    "transformer" = transformer_branch
  )

  representation <- combined %>%
    layer_dense(units = 32, activation = "relu") %>%
    layer_batch_normalization() %>%
    layer_dropout(rate = 0.15)

  propensity_head <- representation %>%
    layer_dense(units = 1, activation = "sigmoid", name = "propensity")
  y0_head <- representation %>%
    layer_dense(units = 1, activation = "linear", name = "Y0")
  y1_head <- representation %>%
    layer_dense(units = 1, activation = "linear", name = "Y1")

  model <- keras_model(inputs = inputs, outputs = list(propensity_head, y0_head, y1_head))

  prop_weight <- ifelse(branch_type == "tarnet", 0.0, 0.5)
  model$compile(
    optimizer = keras_optimizers$Adam(learning_rate = LEARNING_RATE),
    loss = list("binary_crossentropy", "mse", "mse"),
    loss_weights = list(prop_weight, 1.0, 1.0)
  )

  return(model)
}

train_and_predict_dl <- function(branch_type, X_tr_py, Y_tr_list, SW_tr_list,
                                 X_va_py, Y_va_list, SW_va_list, X_te_py,
                                 n_features) {
  mod <- build_temporal_causal_network(n_features = n_features, branch_type = branch_type)
  mod$fit(
    x = X_tr_py, y = Y_tr_list, sample_weight = SW_tr_list,
    validation_data = list(X_va_py, Y_va_list, SW_va_list),
    epochs = as.integer(EPOCHS), batch_size = as.integer(BATCH_SIZE), verbose = 0L
  )
  pred <- mod$predict(X_te_py, verbose = 0L)
  list(
    prop = as.numeric(pred[[1]]),
    y0   = as.numeric(pred[[2]]),
    y1   = as.numeric(pred[[3]]),
    tau  = as.numeric(pred[[3]]) - as.numeric(pred[[2]])
  )
}

###############################################################################
# 5. COMPARATIVE BENCHMARK LEARNERS (S / T / X-LEARNERS)
###############################################################################

fit_meta_learners <- function(train_df, test_df) {
  # S-Learner
  s_fit <- randomForest(Y ~ ., data = train_df, ntree = 150)
  tau_s <- predict(s_fit, newdata = mutate(test_df, A = 1)) -
    predict(s_fit, newdata = mutate(test_df, A = 0))

  # T-Learner
  t_fit_0 <- randomForest(Y ~ . - A, data = filter(train_df, A == 0), ntree = 150)
  t_fit_1 <- randomForest(Y ~ . - A, data = filter(train_df, A == 1), ntree = 150)
  y0_t <- predict(t_fit_0, newdata = test_df)
  y1_t <- predict(t_fit_1, newdata = test_df)
  tau_t <- y1_t - y0_t

  # X-Learner
  tr_1 <- filter(train_df, A == 1)
  tr_0 <- filter(train_df, A == 0)
  d_1 <- tr_1$Y - predict(t_fit_0, newdata = tr_1)
  d_0 <- predict(t_fit_1, newdata = tr_0) - tr_0$Y

  x_fit_1 <- randomForest(x = select(tr_1, -A, -Y), y = d_1, ntree = 150)
  x_fit_0 <- randomForest(x = select(tr_0, -A, -Y), y = d_0, ntree = 150)

  e_fit <- glm(A ~ . - Y, data = train_df, family = binomial)
  e_test <- pmin(pmax(predict(e_fit, newdata = test_df, type = "response"), 0.05), 0.95)

  X_test_cov <- select(test_df, -A, -Y)
  tau_x <- e_test * predict(x_fit_0, newdata = X_test_cov) +
    (1 - e_test) * predict(x_fit_1, newdata = X_test_cov)

  return(list(tau_s = tau_s, tau_t = tau_t, tau_x = tau_x, y0_t = y0_t, y1_t = y1_t))
}

###############################################################################
# 6. POLICY DERIVATION (UNCONSTRAINED & BUDGET-CAPPED) + DR/IPW EVALUATION
###############################################################################

derive_policy <- function(tau_hat, margin = CONTRIBUTION_MARGIN, cost = MARKETING_COST,
                          budget_cap = NULL) {
  net_gain <- (tau_hat * margin) - cost
  eligible <- net_gain > POLICY_THRESHOLD

  if (is.null(budget_cap)) {
    return(as.integer(eligible))
  } else {
    # Target top-K customers among those with positive expected incremental profit
    max_targets <- floor(length(tau_hat) * budget_cap)
    policy <- integer(length(tau_hat))
    if (sum(eligible) > 0 && max_targets > 0) {
      elig_idx <- which(eligible)
      ranked_idx <- elig_idx[order(net_gain[elig_idx], decreasing = TRUE)]
      chosen <- head(ranked_idx, max_targets)
      policy[chosen] <- 1L
    }
    return(policy)
  }
}

# Logged / Observed Campaign Policy Valuation (Doubly Robust & Trimmed IPW)
# Uses nuisance estimates (mu0_nuis, mu1_nuis, prop_nuis) fitted on training folds
evaluate_policy_observed <- function(Y_obs, A_obs, prop_nuis, policy, mu0_nuis, mu1_nuis,
                                     cost = MARKETING_COST, margin = CONTRIBUTION_MARGIN) {
  prop_trim <- pmin(pmax(prop_nuis, 0.05), 0.95)
  m_Y   <- Y_obs * margin
  m_mu0 <- mu0_nuis * margin
  m_mu1 <- mu1_nuis * margin

  mu_pol   <- ifelse(policy == 1, m_mu1 - cost, m_mu0)
  obs_mu   <- ifelse(A_obs == 1, m_mu1, m_mu0)
  obs_prob <- ifelse(A_obs == 1, prop_trim, 1 - prop_trim)

  dr_val  <- mean(mu_pol + (as.numeric(A_obs == policy) / obs_prob) * (m_Y - obs_mu))
  ipw_w   <- pmin(as.numeric(A_obs == policy) / obs_prob, 15)
  ipw_val <- mean(ipw_w * (m_Y - policy * cost))

  return(list(dr = dr_val, ipw = ipw_val))
}

  n_c_valid <- floor(VALID_PROP * N_CUSTOMERS)

  valid_custs <- cust_perm[(n_c_train + 1):(n_c_train + n_c_valid)]
  test_custs  <- cust_perm[(n_c_train + n_c_valid + 1):N_CUSTOMERS]

  train_idx <- which(seq_data$CUST_id %in% train_custs)
  valid_idx <- which(seq_data$CUST_id %in% valid_custs)
  test_idx  <- which(seq_data$CUST_id %in% test_custs)

  stopifnot(length(intersect(seq_data$CUST_id[train_idx], seq_data$CUST_id[test_idx])) == 0)

                     np$array(matrix(Y_tr, ncol = 1), dtype = "float32"))
  SW_tr_list <- list(np$array(rep(1, length(A_tr)), dtype = "float32"),
                     np$array(ifelse(A_tr == 0, 1, 0), dtype = "float32"),
                     np$array(ifelse(A_tr == 1, 1, 0), dtype = "float32"))

  Y_va_list  <- list(np$array(matrix(A_va, ncol = 1), dtype = "float32"),
                     np$array(matrix(Y_va, ncol = 1), dtype = "float32"),
                     np$array(matrix(Y_va, ncol = 1), dtype = "float32"))
  SW_va_list <- list(np$array(rep(1, length(A_va)), dtype = "float32"),
                     np$array(ifelse(A_va == 0, 1, 0), dtype = "float32"),
                     np$array(ifelse(A_va == 1, 1, 0), dtype = "float32"))

  res_bilstm <- train_and_predict_dl("bilstm", X_tr_py, Y_tr_list, SW_tr_list, X_va_py, Y_va_list, SW_va_list, X_te_py, n_features_total)
  res_trans  <- train_and_predict_dl("transformer", X_tr_py, Y_tr_list, SW_tr_list, X_va_py, Y_va_list, SW_va_list, X_te_py, n_features_total)
  res_tarnet <- train_and_predict_dl("tarnet", X_tr_py, Y_tr_list, SW_tr_list, X_va_py, Y_va_list, SW_va_list, X_te_py, n_features_total)

  # 3A. Fit Meta-Learners on EQUAL Full Flattened Sequence Features
  train_flat_df <- as.data.frame(seq_data$FLAT_seq[train_idx, ])
  train_flat_df$A <- A_tr; train_flat_df$Y <- Y_tr
  test_flat_df  <- as.data.frame(seq_data$FLAT_seq[test_idx, ])
  test_flat_df$A <- A_te; test_flat_df$Y <- Y_te
  bench_seq <- fit_meta_learners(train_flat_df, test_flat_df)

  # 3B. Fit Meta-Learners on Static RFM Summaries (Feature Representation Comparison)
  train_rfm_df <- as.data.frame(seq_data$RFM_seq[train_idx, ])
  train_rfm_df$A <- A_tr; train_rfm_df$Y <- Y_tr
  test_rfm_df  <- as.data.frame(seq_data$RFM_seq[test_idx, ])
  test_rfm_df$A <- A_te; test_rfm_df$Y <- Y_te
  bench_rfm <- fit_meta_learners(train_rfm_df, test_rfm_df)

  # 4. Derive Policies (Unconstrained Net-Profit Rule)
  pol_full      <- derive_policy(res_full$tau)
  pol_cnn       <- derive_policy(res_cnn$tau)
  pol_bilstm    <- derive_policy(res_bilstm$tau)
  pol_trans     <- derive_policy(res_trans$tau)
  pol_tarnet    <- derive_policy(res_tarnet$tau)
  pol_s_seq     <- derive_policy(bench_seq$tau_s)
  pol_t_seq     <- derive_policy(bench_seq$tau_t)
  pol_x_seq     <- derive_policy(bench_seq$tau_x)
  pol_x_rfm     <- derive_policy(bench_rfm$tau_x)

  # Predictive Targeting (Top-K by predicted purchase revenue Y1 matching proposed rate)
  target_rate <- mean(pol_full)
  pred_cutoff <- quantile(res_full$y1, probs = 1 - target_rate)
  pol_pred    <- as.integer(res_full$y1 >= pred_cutoff)

  # Random Targeting (matched targeting rate) & Oracle Policy
  pol_random  <- rbinom(length(test_idx), 1, target_rate)
  true_tau_te <- seq_data$TAU_seq[test_idx]
  true_y0_te  <- seq_data$Y0_seq[test_idx]
  true_y1_te  <- seq_data$Y1_seq[test_idx]
  pol_oracle  <- derive_policy(true_tau_te)
  pol_all     <- rep(1L, length(test_idx))
  pol_none    <- rep(0L, length(test_idx))

  # 5. Evaluate Ground-Truth Profit & Out-of-Sample Observed DR/IPW Profit
  true_profit <- function(pol) {
    mean(ifelse(pol == 1, (true_y1_te * CONTRIBUTION_MARGIN) - MARKETING_COST,
                true_y0_te * CONTRIBUTION_MARGIN))
  }

  # Use independent T-Learner nuisance estimates for DR/IPW logged evaluation
  obs_eval_full <- evaluate_policy_observed(
    Y_te, A_te, res_full$prop, pol_full, bench_seq$y0_t, bench_seq$y1_t
  )
  obs_eval_x <- evaluate_policy_observed(
    Y_te, A_te, res_full$prop, pol_x_seq, bench_seq$y0_t, bench_seq$y1_t
  )
  obs_eval_pred <- evaluate_policy_observed(
    Y_te, A_te, res_full$prop, pol_pred, bench_seq$y0_t, bench_seq$y1_t
  )

  pehe <- function(est) sqrt(mean((est - true_tau_te)^2))

  mc_results_list[[mc]] <- data.frame(
    Replication      = mc,
    # PEHE (CATE Estimation Accuracy)
    PEHE_Full        = pehe(res_full$tau),
    PEHE_CNN         = pehe(res_cnn$tau),
    PEHE_BiLSTM      = pehe(res_bilstm$tau),
    PEHE_Transformer = pehe(res_trans$tau),
    PEHE_TARNet      = pehe(res_tarnet$tau),
    PEHE_SLearnerSeq = pehe(bench_seq$tau_s),
    PEHE_TLearnerSeq = pehe(bench_seq$tau_t),
    PEHE_XLearnerSeq = pehe(bench_seq$tau_x),
    PEHE_XLearnerRFM = pehe(bench_rfm$tau_x),
    # Ground-Truth Average Profit per Customer ($)
    Profit_Oracle      = true_profit(pol_oracle),
    Profit_Full        = true_profit(pol_full),
    Profit_CNN         = true_profit(pol_cnn),
    Profit_BiLSTM      = true_profit(pol_bilstm),
    Profit_Transformer = true_profit(pol_trans),
    Profit_TARNet      = true_profit(pol_tarnet),
    Profit_SLearnerSeq = true_profit(pol_s_seq),
    Profit_TLearnerSeq = true_profit(pol_t_seq),
    Profit_XLearnerSeq = true_profit(pol_x_seq),
    Profit_XLearnerRFM = true_profit(pol_x_rfm),
    Profit_Predictive  = true_profit(pol_pred),
    Profit_Random      = true_profit(pol_random),
    Profit_TargetAll   = true_profit(pol_all),
    Profit_TargetNone  = true_profit(pol_none),
    # Observed/Logged Policy Evaluation (DR & IPW)
    DR_Profit_Full       = obs_eval_full$dr,
    DR_Profit_XLearner   = obs_eval_x$dr,
    DR_Profit_Predictive = obs_eval_pred$dr,
    IPW_Profit_Full      = obs_eval_full$ipw,
    # Targeting Rates
    Rate_Oracle = mean(pol_oracle),
    Rate_Full   = target_rate
  )

  # 6. Budget-Constrained Sensitivity Analysis (Reviewer #2, Point 5)
  for (b_cap in BUDGET_CAPS) {
    budget_results_list[[length(budget_results_list) + 1]] <- data.frame(
      Replication = mc,
      Budget_Cap  = b_cap,
      Oracle      = true_profit(derive_policy(true_tau_te, budget_cap = b_cap)),
      Full_DL     = true_profit(derive_policy(res_full$tau, budget_cap = b_cap)),
      XLearner    = true_profit(derive_policy(bench_seq$tau_x, budget_cap = b_cap)),
      Predictive  = true_profit(as.integer(res_full$y1 >= quantile(res_full$y1, 1 - b_cap))),
      Random      = true_profit(rbinom(length(test_idx), 1, b_cap))
    )
  }
}

mc_results_df     <- do.call(rbind, mc_results_list)
budget_results_df <- do.call(rbind, budget_results_list)

compute_paired_stats <- function(diff_vec, name) {
  m  <- mean(diff_vec)
  se <- sd(diff_vec) / sqrt(length(diff_vec))
  ci_low  <- m - 1.96 * se
  ci_high <- m + 1.96 * se
  p_val   <- t.test(diff_vec, mu = 0)$p.value
  data.frame(
    Comparison = name, Mean_Diff = m, SE = se,
    CI_95_Low = ci_low, CI_95_High = ci_high, P_Value = p_val
  )
}

paired_table <- rbind(
  compute_paired_stats(mc_results_df$Profit_Full - mc_results_df$Profit_XLearnerSeq, "Full DL vs. X-Learner (Seq)"),
  compute_paired_stats(mc_results_df$Profit_Full - mc_results_df$Profit_XLearnerRFM, "Full DL vs. X-Learner (RFM)"),
  compute_paired_stats(mc_results_df$Profit_Full - mc_results_df$Profit_TLearnerSeq, "Full DL vs. T-Learner (Seq)"),
  compute_paired_stats(mc_results_df$Profit_Full - mc_results_df$Profit_TARNet,      "Full DL vs. TARNet (Ablation)"),
  compute_paired_stats(mc_results_df$Profit_Full - mc_results_df$Profit_Predictive,  "Full DL vs. Predictive Targeting"),
  compute_paired_stats(mc_results_df$Profit_Full - mc_results_df$Profit_TargetNone,  "Full DL vs. No Intervention"),
  compute_paired_stats(mc_results_df$Profit_Oracle - mc_results_df$Profit_Full,      "Oracle Gap (Oracle - Full DL)")
)
